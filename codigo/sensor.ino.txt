// Clasificación del estado de una tapa con KY-032

const int SENSOR = 2;

void setup() {
  Serial.begin(9600);
  pinMode(SENSOR, INPUT);
}

void loop() {
  int estado = digitalRead(SENSOR);

  if (estado == LOW) {
    Serial.println("Tapa detectada - CERRADA");
  } else {
    Serial.println("Tapa no detectada - SIN TAPA");
  }

  delay(500);
}
